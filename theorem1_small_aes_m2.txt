static uint32_t rng_state[4];

static inline uint32_t rotl(const uint32_t x, int k)
{
    return (x << k) | (x >> (32 - k));
}

static uint32_t next_prng(void)
{
    const uint32_t result =
        rotl(rng_state[0] + rng_state[3], 7) + rng_state[0];

    const uint32_t t = rng_state[1] << 9;

    rng_state[2] ^= rng_state[0];
    rng_state[3] ^= rng_state[1];
    rng_state[1] ^= rng_state[2];
    rng_state[0] ^= rng_state[3];

    rng_state[2] ^= t;
    rng_state[3] = rotl(rng_state[3], 11);

    return result;
}

static void init_prng(uint32_t seed)
{
    rng_state[0] = seed ^ 0x12345678;
    rng_state[1] = seed ^ 0x87654321;
    rng_state[2] = seed ^ 0xDEADBEEF;
    rng_state[3] = seed ^ 0xCAFEBABE;

    for (int i = 0; i < 20; ++i)
        next_prng();
}

static const uint8_t SMALL_AES_SBOX[16] = {
    0x6, 0xB, 0x5, 0x4,
    0x2, 0xE, 0x7, 0xA,
    0x9, 0xD, 0xF, 0xC,
    0x3, 0x1, 0x0, 0x8
};

static uint8_t gf16_mul(uint8_t a, uint8_t b)
{
    uint8_t result = 0;

    a &= 0xF;
    b &= 0xF;

    for (int i = 0; i < 4; ++i)
    {
        if (b & 1)
            result ^= a;

        b >>= 1;

        uint8_t carry = a & 0x8;

        a <<= 1;

        if (carry)
            a ^= 0x13;

        a &= 0xF;
    }

    return result & 0xF;
}

static uint8_t MUL2[16], MUL3[16];
static void init_gf_tables(void) {
    for (int x=0;x<16;++x) {
        MUL2[x]=gf16_mul(2,(uint8_t)x);
        MUL3[x]=gf16_mul(3,(uint8_t)x);
    }
}

static void sub_bytes(uint8_t state[STATE_NIBBLES])
{
    for (int i = 0; i < STATE_NIBBLES; ++i)
        state[i] = SMALL_AES_SBOX[state[i] & 0xF];
}

static void shift_rows(uint8_t state[STATE_NIBBLES])
{
    uint8_t tmp[STATE_NIBBLES];

    tmp[0]  = state[0];
    tmp[1]  = state[5];
    tmp[2]  = state[10];
    tmp[3]  = state[15];

    tmp[4]  = state[4];
    tmp[5]  = state[9];
    tmp[6]  = state[14];
    tmp[7]  = state[3];

    tmp[8]  = state[8];
    tmp[9]  = state[13];
    tmp[10] = state[2];
    tmp[11] = state[7];

    tmp[12] = state[12];
    tmp[13] = state[1];
    tmp[14] = state[6];
    tmp[15] = state[11];

    memcpy(state, tmp, STATE_NIBBLES);
}

static void mix_columns(uint8_t state[STATE_NIBBLES])
{
    uint8_t tmp[STATE_NIBBLES];

    for (int col = 0; col < 4; ++col)
    {
        int base = 4 * col;

        uint8_t c0 = state[base + 0] & 0xF;
        uint8_t c1 = state[base + 1] & 0xF;
        uint8_t c2 = state[base + 2] & 0xF;
        uint8_t c3 = state[base + 3] & 0xF;

        tmp[base + 0] =
            MUL2[c0] ^
            MUL3[c1] ^
            c2 ^
            c3;

        tmp[base + 1] =
            c0 ^
            MUL2[c1] ^
            MUL3[c2] ^
            c3;

        tmp[base + 2] =
            c0 ^
            c1 ^
            MUL2[c2] ^
            MUL3[c3];

        tmp[base + 3] =
            MUL3[c0] ^
            c1 ^
            c2 ^
            MUL2[c3];

        tmp[base + 0] &= 0xF;
        tmp[base + 1] &= 0xF;
        tmp[base + 2] &= 0xF;
        tmp[base + 3] &= 0xF;
    }

    memcpy(state, tmp, STATE_NIBBLES);
}

static void add_round_key(
    uint8_t state[STATE_NIBBLES],
    const uint8_t RK[STATE_NIBBLES])
{
    for (int i = 0; i < STATE_NIBBLES; ++i)
        state[i] ^= RK[i] & 0xF;
}

static void small_aes_key_schedule(
    const uint8_t key[KEY_BYTES],
    uint8_t round_keys[ROUNDS + 1][STATE_NIBBLES])
{
    
    for (int i = 0; i < KEY_BYTES; ++i)
    {
        round_keys[0][2 * i] =
            (key[i] >> 4) & 0xF;

        round_keys[0][2 * i + 1] =
            key[i] & 0xF;
    }

    uint8_t kappa = 0x1;

    for (int r = 1; r <= ROUNDS; ++r)
    {
        const uint8_t *prev = round_keys[r - 1];
        uint8_t *curr = round_keys[r];

        uint8_t temp[4];

        temp[0] = SMALL_AES_SBOX[prev[13]];
        temp[1] = SMALL_AES_SBOX[prev[14]];
        temp[2] = SMALL_AES_SBOX[prev[15]];
        temp[3] = SMALL_AES_SBOX[prev[12]];

        temp[0] ^= kappa;

        for (int j = 0; j < 4; ++j)
            curr[j] = (prev[j] ^ temp[j]) & 0xF;

        for (int q = 1; q < 4; ++q)
        {
            for (int j = 0; j < 4; ++j)
            {
                curr[4 * q + j] =
                    (prev[4 * q + j] ^
                     curr[4 * (q - 1) + j]) & 0xF;
            }
        }

        kappa = gf16_mul(0x2, kappa);
    }
}

static void small_aes_round(
    uint8_t state[STATE_NIBBLES],
    const uint8_t RK[STATE_NIBBLES])
{
    sub_bytes(state);
    shift_rows(state);
    mix_columns(state);
    add_round_key(state, RK);
}

static void small_aes_final_round(
    uint8_t state[STATE_NIBBLES],
    const uint8_t RK[STATE_NIBBLES])
{
    sub_bytes(state);
    shift_rows(state);
    add_round_key(state, RK);
}

static void small_aes_encrypt(
    uint8_t state[STATE_NIBBLES],
    const uint8_t round_keys[ROUNDS + 1][STATE_NIBBLES])
{
    add_round_key(state, round_keys[0]);

    for (int r = 1; r < ROUNDS; ++r)
        small_aes_round(state, round_keys[r]);

    small_aes_final_round(state, round_keys[ROUNDS]);
}

#define TEXTS_PER_STRUCTURE 256
#define PAIRS_PER_STRUCTURE UINT64_C(28800)
#ifndef ACTIVE_A
#define ACTIVE_A 0
#endif
#ifndef ACTIVE_B
#define ACTIVE_B 5
#endif
#if ACTIVE_A < 0 || ACTIVE_A > 15 || ACTIVE_B < 0 || ACTIVE_B > 15 || ACTIVE_A == ACTIVE_B
#error Invalid active positions
#endif
#if STRUCTURES_LOG2 < 2 || STRUCTURES_LOG2 > 56
#error Invalid structure exponent
#endif
#define NUM_COUNTERS         15

static void generate_fixed_part(uint8_t fixed[BLOCK_NIBBLES],
                                uint64_t index, uint64_t mul, uint64_t offset)
{
    const uint64_t mask = (UINT64_C(1)<<56)-1;
    uint64_t x = (index + offset) & mask;
    x ^= x >> 27; x = (x * mul) & mask;
    x ^= x >> 23; x = (x * UINT64_C(0x9e3779b97f4a7c15)) & mask;
    x ^= x >> 29;
    for (int i=0;i<16;++i) {
        if (i==ACTIVE_A || i==ACTIVE_B) fixed[i]=0;
        else { fixed[i]=(uint8_t)(x & 15); x >>= 4; }
    }
}

static void generate_plaintext(
    uint8_t plaintext[BLOCK_NIBBLES],
    const uint8_t fixed[BLOCK_NIBBLES],
    uint8_t active_value)
{
    memcpy(plaintext, fixed, BLOCK_NIBBLES);

    plaintext[ACTIVE_A] = active_value >> 4;
    plaintext[ACTIVE_B] = active_value & 0xF;
}

static inline uint8_t get_passive_inverse_diagonal_mask(
    const uint8_t c1[BLOCK_NIBBLES],
    const uint8_t c2[BLOCK_NIBBLES])
{
    uint8_t mask = 0;

    if (((c1[0]  ^ c2[0])  == 0) &&
        ((c1[7]  ^ c2[7])  == 0) &&
        ((c1[10] ^ c2[10]) == 0) &&
        ((c1[13] ^ c2[13]) == 0))
    {
        mask |= 0x1;
    }

    if (((c1[1]  ^ c2[1])  == 0) &&
        ((c1[4]  ^ c2[4])  == 0) &&
        ((c1[11] ^ c2[11]) == 0) &&
        ((c1[14] ^ c2[14]) == 0))
    {
        mask |= 0x2;
    }

    if (((c1[2]  ^ c2[2])  == 0) &&
        ((c1[5]  ^ c2[5])  == 0) &&
        ((c1[8]  ^ c2[8])  == 0) &&
        ((c1[15] ^ c2[15]) == 0))
    {
        mask |= 0x4;
    }

    if (((c1[3]  ^ c2[3])  == 0) &&
        ((c1[6]  ^ c2[6])  == 0) &&
        ((c1[9]  ^ c2[9])  == 0) &&
        ((c1[12] ^ c2[12]) == 0))
    {
        mask |= 0x8;
    }

    return mask;
}

static uint64_t ROUND_T[16][16];
static inline uint64_t pack_state(const uint8_t a[16]) {
    uint64_t x=0;
    for (int i=0;i<16;++i) x|=(uint64_t)(a[i]&15)<<(4*i);
    return x;
}
static inline void unpack_state(uint64_t x,uint8_t a[16]) {
    for(int i=0;i<16;++i) a[i]=(uint8_t)((x>>(4*i))&15);
}
static void init_round_tables(void) {
    for(int i=0;i<16;++i) for(int x=0;x<16;++x) {
        uint8_t a[16]={0};
        a[i]=SMALL_AES_SBOX[x];
        shift_rows(a);
        mix_columns(a);
        ROUND_T[i][x]=pack_state(a);
    }
}
static inline uint64_t fast_round(uint64_t s,uint64_t key) {
    uint64_t t=key;
    for(int i=0;i<16;++i)
        t ^= ROUND_T[i][(s>>(4*i))&15];
    return t;
}
static inline uint64_t fast_pre_final(uint64_t s,const uint64_t rk[ROUNDS+1]) {
    s ^= rk[0];
    for(int r=1;r<ROUNDS;++r) s=fast_round(s,rk[r]);
    return s;
}

static inline void pre_final_to_virtual_ciphertext(uint64_t s,uint8_t ct[16]) {
    static const uint8_t pos[4][4]={{0,7,10,13},{1,4,11,14},{2,5,8,15},{3,6,9,12}};
    for(int d=0;d<4;++d)
        for(int j=0;j<4;++j)
            ct[pos[d][j]]=(uint8_t)((s>>(4*(4*d+j)))&15);
}
static int validate_fast_cipher(void) {
    uint8_t key[8],rk[ROUNDS+1][16],plain[16],a[16],b[16];
    uint64_t prk[ROUNDS+1];
    for(int k=0;k<8;++k) {
        for(int i=0;i<8;++i) key[i]=(uint8_t)((k*31+i*73+11)&255);
        small_aes_key_schedule(key,rk);
        for(int r=0;r<=ROUNDS;++r) prk[r]=pack_state(rk[r]);
        for(int v=0;v<256;++v) {
            for(int i=0;i<16;++i) plain[i]=(uint8_t)((v*13+i*7+k*3)&15);
            memcpy(a,plain,16); small_aes_encrypt(a,rk);
            pre_final_to_virtual_ciphertext(fast_pre_final(pack_state(plain),prk),b);
            
            uint8_t plain2[16],a2[16],b2[16];
            for(int i=0;i<16;++i) plain2[i]=(uint8_t)((v*9+i*11+k*5+1)&15);
            memcpy(a2,plain2,16); small_aes_encrypt(a2,rk);
            pre_final_to_virtual_ciphertext(fast_pre_final(pack_state(plain2),prk),b2);
            if(get_passive_inverse_diagonal_mask(a,a2)!=get_passive_inverse_diagonal_mask(b,b2)) return 0;
        }
    }
    return 1;
}

static inline int popcount4(uint8_t x)
{
    return
        ((x >> 0) & 1) +
        ((x >> 1) & 1) +
        ((x >> 2) & 1) +
        ((x >> 3) & 1);
}

enum
{
    C_D0 = 0,
    C_D1,
    C_D2,
    C_D3,

    C_D0_D1,
    C_D0_D2,
    C_D0_D3,
    C_D1_D2,
    C_D1_D3,
    C_D2_D3,

    C_EXACTLY_1_PASSIVE,
    C_EXACTLY_2_PASSIVE,

    C_AT_LEAST_1_PASSIVE,
    C_AT_LEAST_2_PASSIVE,
    C_EXACTLY_3_PASSIVE
};

static const char *COUNTER_NAMES[NUM_COUNTERS] =
{
    "D0 passive; D1,D2,D3 free",
    "D1 passive; D0,D2,D3 free",
    "D2 passive; D0,D1,D3 free",
    "D3 passive; D0,D1,D2 free",

    "D0,D1 passive; D2,D3 free",
    "D0,D2 passive; D1,D3 free",
    "D0,D3 passive; D1,D2 free",
    "D1,D2 passive; D0,D3 free",
    "D1,D3 passive; D0,D2 free",
    "D2,D3 passive; D0,D1 free",

    "Exactly 1 passive + 3 active",
    "Exactly 2 passive + 2 active",

    "At least 1 passive",
    "At least 2 passive",
    "Exactly 3 passive + 1 active"
};

static inline void update_counters(
    uint64_t counters[NUM_COUNTERS],
    uint8_t mask)
{
    int number_of_passive =
        popcount4(mask);

    if (mask & 0x1)
        ++counters[C_D0];

    if (mask & 0x2)
        ++counters[C_D1];

    if (mask & 0x4)
        ++counters[C_D2];

    if (mask & 0x8)
        ++counters[C_D3];

    if ((mask & 0x3) == 0x3)
        ++counters[C_D0_D1];

    if ((mask & 0x5) == 0x5)
        ++counters[C_D0_D2];

    if ((mask & 0x9) == 0x9)
        ++counters[C_D0_D3];

    if ((mask & 0x6) == 0x6)
        ++counters[C_D1_D2];

    if ((mask & 0xA) == 0xA)
        ++counters[C_D1_D3];

    if ((mask & 0xC) == 0xC)
        ++counters[C_D2_D3];

    if (number_of_passive == 1)
        ++counters[C_EXACTLY_1_PASSIVE];

    if (number_of_passive == 2)
        ++counters[C_EXACTLY_2_PASSIVE];

    if (number_of_passive == 3)
        ++counters[C_EXACTLY_3_PASSIVE];

    if (number_of_passive >= 1)
        ++counters[C_AT_LEAST_1_PASSIVE];

    if (number_of_passive >= 2)
        ++counters[C_AT_LEAST_2_PASSIVE];
}

#ifndef BRUTE_FORCE
#define BRUTE_FORCE 0
#endif
static int valid_pair(int a,int b) {
    unsigned d=(unsigned)(a^b);
    return (d&0xf0)!=0 && (d&15)!=0;
}
static void brute(const uint8_t ct[256][16], uint64_t c[NUM_COUNTERS]) {
    for(int a=0;a<256;++a) for(int b=a+1;b<256;++b)
        if(valid_pair(a,b)) update_counters(c,get_passive_inverse_diagonal_mask(ct[a],ct[b]));
}

static int cmp_u32(const void *a,const void *b) {
    uint32_t x=*(const uint32_t*)a,y=*(const uint32_t*)b;
    return (x>y)-(x<y);
}
static void grouped(const uint8_t ct[256][16],uint64_t c[NUM_COUNTERS]) {
    static const unsigned pos[4][4]={{0,7,10,13},{1,4,11,14},{2,5,8,15},{3,6,9,12}};
    for (unsigned d=0;d<4;++d) {
        uint32_t keys[512];
        int16_t head[512], next[256];
        for (int z=0;z<512;++z) head[z]=-1;
        for (unsigned a=0;a<256;++a) {
            unsigned v=0;
            for(unsigned j=0;j<4;++j) v=(v<<4)|ct[a][pos[d][j]];
            unsigned slot=(v*2654435761u)>>23;
            while(head[slot]!=-1 && keys[slot]!=v) slot=(slot+1)&511;
            if(head[slot]==-1) {keys[slot]=v;next[a]=-1;head[slot]=(int16_t)a;}
            else {
                for(int b=head[slot];b!=-1;b=next[b]) {
                    if(!valid_pair((int)a,b)) continue;
                    uint8_t mask=get_passive_inverse_diagonal_mask(ct[a],ct[b]);
                    if(mask & ((1u<<d)-1)) continue;
                    update_counters(c,mask);
                }
                next[a]=head[slot];head[slot]=(int16_t)a;
            }
        }
    }
}
static void report(const char *label,const uint64_t c[NUM_COUNTERS],uint64_t n) {
    printf("\n%s | Pairs=%" PRIu64 "\n",label,n);
    for(int i=0;i<18;++i) {
        const char *name;
        uint64_t count;
        if(i==0) {name="0 specified passive; all free";count=n;}
        else if(i==11) {name="Exactly 0 passive + 4 active";count=n-c[C_AT_LEAST_1_PASSIVE];}
        else if(i==14) {name="At least 0 passive";count=n;}
        else if(i==17) {name=COUNTER_NAMES[C_EXACTLY_3_PASSIVE];count=c[C_EXACTLY_3_PASSIVE];}
        else {int j=i-(i>0)-(i>11)-(i>14);name=COUNTER_NAMES[j];count=c[j];}
        double p=(double)count/(double)n;
        printf("%2d. %-38s count=%" PRIu64 " p=%.17g log2(p)=%.15f\n",i+1,name,count,p,count?log2(p):-INFINITY);
    }
    fflush(stdout);
}
static int validate_counting(void) {
    uint8_t key[8]={0},rk[ROUNDS+1][16],fixed[16],ct[256][16];
    small_aes_key_schedule(key,rk);
    for(unsigned j=0;j<64;++j) {
        uint64_t a[NUM_COUNTERS]={0},b[NUM_COUNTERS]={0};
        generate_fixed_part(fixed,j,UINT64_C(0x123456789abcdef1),123);
        for(unsigned v=0;v<256;++v) {
            generate_plaintext(ct[v],fixed,(uint8_t)v);
            small_aes_encrypt(ct[v],rk);
        }
        brute(ct,a); grouped(ct,b);
        if(memcmp(a,b,sizeof a)) return 0;
    }
    return 1;
}
static int validate_structure(void) {
    uint8_t fixed[16], pt[256][16];
    generate_fixed_part(fixed,0,1,0);
    for(int a=0;a<256;++a) generate_plaintext(pt[a],fixed,(uint8_t)a);
    unsigned accepted=0;
    for(int a=0;a<256;++a) for(int b=a+1;b<256;++b) {
        unsigned mask=0;
        for(int i=0;i<16;++i) if(pt[a][i]!=pt[b][i]) mask |= 1u<<i;
        int valid=((a>>4)!=(b>>4) && (a&15)!=(b&15));
        if(valid != (mask==((1u<<ACTIVE_A)|(1u<<ACTIVE_B)))) return 0;
        accepted+=valid;
    }
    return accepted==PAIRS_PER_STRUCTURE;
}

int main(void)
{
    init_gf_tables();
    init_round_tables();
    if(!validate_fast_cipher()) { fprintf(stderr,"Packed cipher self-test failed\n"); return 1; }
    printf("Packed cipher self-test passed: 8 keys x 256 paired plaintexts\n");
    if(!validate_structure()) { fprintf(stderr,"Structure self-test failed\n"); return 1; }
    printf("Structure self-test passed: 28800 exact m=2 pairs\n");
    if(!validate_counting()) { fprintf(stderr,"Grouping self-test failed\n"); return 1; }
    printf("Grouping self-test passed: 64 structures vs brute-force\n");
    printf("Counting method: %s (packed T-table / 4 rounds)\n",BRUTE_FORCE?"brute-force":"grouped");
    uint32_t seed=RNG_SEED;
    init_prng(seed);
    omp_set_dynamic(0); omp_set_num_threads(NUM_THREADS);
    #pragma omp parallel
    {
        #pragma omp single
        printf("OpenMP threads: %d | seed=%" PRIu32 "\n",omp_get_num_threads(),seed);
    }
    if (NUM_KEYS <= 0 || NUM_STRUCTURES > UINT64_MAX/PAIRS_PER_STRUCTURE/(uint64_t)NUM_KEYS) { fprintf(stderr,"Pair count overflow\n"); return 1; }
    printf("m=2 | Active positions: %d,%d | Texts/structure=256 | Valid pairs/structure=28800\n",ACTIVE_A,ACTIVE_B);
    const uint64_t pairs_per_key=NUM_STRUCTURES*PAIRS_PER_STRUCTURE;
    uint64_t total[NUM_COUNTERS]={0};
    printf("Small-AES rounds=%d keys=%d structures/key=%" PRIu64 " pairs/key=%" PRIu64 "\n",ROUNDS,NUM_KEYS,NUM_STRUCTURES,pairs_per_key);
    for(int k=0;k<NUM_KEYS;++k) {
        uint8_t key[KEY_BYTES], rk[ROUNDS+1][STATE_NIBBLES];
        for(int i=0;i<KEY_BYTES;++i) key[i]=(uint8_t)next_prng();
        small_aes_key_schedule(key,rk);
        uint64_t packed_rk[ROUNDS+1];
        for(int r=0;r<=ROUNDS;++r) packed_rk[r]=pack_state(rk[r]);
        uint64_t mul=(uint64_t)next_prng()<<32; mul|=next_prng(); mul|=1;
        uint64_t offset=(uint64_t)next_prng()<<32; offset|=next_prng();
        uint64_t counts[NUM_COUNTERS]={0}, tested=0;
        double start=omp_get_wtime();
        printf("KEY %d/%d | master key=",k+1,NUM_KEYS);
        for(int i=0;i<KEY_BYTES;++i) printf("%02x",(unsigned)key[i]);
        printf("\n"); fflush(stdout);
        for(unsigned stage=0;stage<4;++stage) {
            uint64_t begin=NUM_STRUCTURES*stage/4, end=NUM_STRUCTURES*(stage+1)/4;
            #pragma omp parallel
            {
                uint64_t local[NUM_COUNTERS]={0}, local_pairs=0;
                #pragma omp for schedule(static)
                for(uint64_t j=begin;j<end;++j) {
                    uint8_t fixed[16], ct[256][16];
                    generate_fixed_part(fixed,j,mul,offset);
                    for(int v=0;v<256;++v) {
                        generate_plaintext(ct[v],fixed,(uint8_t)v);
                        pre_final_to_virtual_ciphertext(fast_pre_final(pack_state(ct[v]),packed_rk),ct[v]);
                    }
                    if(BRUTE_FORCE) brute(ct,local); else grouped(ct,local);
                    local_pairs+=PAIRS_PER_STRUCTURE;
                }
                #pragma omp critical
                { for(int i=0;i<NUM_COUNTERS;++i) counts[i]+=local[i]; tested+=local_pairs; }
            }
            if(tested != end*PAIRS_PER_STRUCTURE) { fprintf(stderr,"Pair count mismatch\n"); return 1; }
            double elapsed=omp_get_wtime()-start;
            printf("KEY %d/%d | Progress: %u%% | Structures=%" PRIu64 "/%" PRIu64 " | Elapsed=%.1f s | ETA=%.1f s\n",k+1,NUM_KEYS,25*(stage+1),end,NUM_STRUCTURES,elapsed,elapsed*(3-stage)/(stage+1));
            fflush(stdout);
        }
        if (counts[C_AT_LEAST_1_PASSIVE] > pairs_per_key || counts[C_AT_LEAST_1_PASSIVE] != counts[C_EXACTLY_1_PASSIVE]+counts[C_AT_LEAST_2_PASSIVE] || counts[C_AT_LEAST_2_PASSIVE]<counts[C_EXACTLY_2_PASSIVE]) { fprintf(stderr,"Counter consistency failure\n"); return 1; }
        for(int i=0;i<NUM_COUNTERS;++i) total[i]+=counts[i];
        char label[64]; snprintf(label,sizeof label,"KEY %d FINISHED",k+1);
        report(label,counts,pairs_per_key);
    }
    report("ALL KEYS FINISHED",total,pairs_per_key*(uint64_t)NUM_KEYS);
    return 0;
}
